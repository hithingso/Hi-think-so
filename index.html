// 頁面切換邏輯

function switchPage(pageId) {

document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));

document.querySelectorAll('nav a').forEach(a => a.classList.remove('active'));

 

document.getElementById('page-' + pageId).classList.add('active');

document.getElementById('nav-' + pageId).classList.add('active');

window.scrollTo(0, 0);

}

 

// 情緒罐互動

function openMoodModal() {

const note = prompt("今天發生了什麼事？或想寫下一小段話：");

if (note) {

const colors = ['#E3A857', '#7C9D96', '#5B82A6', '#D48372', '#9B82C1'];

const randomColor = colors[Math.floor(Math.random() * colors.length)];

const jar = document.getElementById('jarContainer');

const ball = document.createElement('div');

ball.className = 'mood-ball';

ball.style.background = randomColor;

ball.title = note;

ball.innerText = "●";

jar.appendChild(ball);

alert("成功放入情緒罐！");

}

}

 

// AI 聊天模擬

function sendQuick(text) {

appendMsg(text, 'user');

setTimeout(() => {

let reply = "我聽到了。謝謝你願意跟我分享這件事，這一定不容易。想多聊聊當時是什麼情況嗎？";

if(text.includes('壓力')) reply = "壓力大時記得給自己喘息的空間。你最近有試著做點讓自己放鬆的小事嗎？";

appendMsg(reply, 'ai');

}, 600);

}

 

function sendMessage() {

const input = document.getElementById('userInput');

if (!input.value.trim()) return;

const text = input.value;

appendMsg(text, 'user');

input.value = '';

 

setTimeout(() => {

appendMsg("我懂你的感受。在這裡你可以放輕鬆，不管發生什麼事，都有人願意聽你說。如果有需要，我們也可以看看「找幫助」裡的資源喔！", 'ai');

}, 600);

}

 

function appendMsg(text, sender) {

const box = document.getElementById('chatBox');

const msg = document.createElement('div');

msg.className = `msg ${sender}`;

msg.innerText = text;

box.appendChild(msg);

box.scrollTop = box.scrollHeight;

}

 

// 小遊戲互動解析

function selectOption(opt) {

const resultBox = document.getElementById('gameResult');

resultBox.style.display = 'block';

if (opt === 3) {

resultBox.innerHTML = "<strong>✨ 很棒的選擇！解析：</strong><br>每個人都有自己的步調與生活忙碌的時候。給彼此空間，不僅能避免過度焦慮，也能讓對話在彼此都舒服的狀態下進行。";

resultBox.style.borderLeft = "4px solid #7C9D96";

} else {

resultBox.innerHTML = "<strong>💡 小建議：</strong><br>過度急切或直接把不安丟給對方，有時會造成雙方的壓力。試著先照顧自己的情緒，給對方一些時間，通常會有更好的互動結果喔！";

resultBox.style.borderLeft = "4px solid #E3A857";

}

}

 

// 資源篩選

function filterResource(cat) {

const items = document.querySelectorAll('.map-box');

items.forEach(item => {

if (cat === 'all' || item.dataset.category === cat) {

item.style.display = 'block';

} else {

item.style.display = 'none';

}

});

}
